function highlight(json) {
return JSON.stringify(json, null, 2)
.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
.replace(/("(?:\\u[a-fA-F0-9]{4}|\\[^u]|[^\\"])*"(\s*:)?|\b(?:true|false|null)\b|-?\d+(?:\.\d*)?(?:[eE][+-]?\d+)?)/g, m => {
if (/^"/.test(m)) return /:$/.test(m) ? `<span class="jk">${m}</span>` : `<span class="js">${m}</span>`;
if (/true|false/.test(m)) return `<span class="jb">${m}</span>`;
if (/null/.test(m)) return `<span class="jl">${m}</span>`;
return `<span class="jn">${m}</span>`;
});
}

const carouselTracks = Array.from(document.querySelectorAll('.carousel-track'));
const carouselTimers = new WeakMap();
const carouselRotationInterval = 3000;
let carouselAutoRotateEnabled = false;

function getCarouselSlides(track) {
return Array.from(track.children).filter(slide => !slide.dataset.loopClone);
}

function getCarouselIndex(track) {
const count = Number(track.dataset.realSlideCount || getCarouselSlides(track).length);
if (!count) return 0;
return (Math.round(track.scrollLeft / track.clientWidth) - 1 + count) % count;
}

function setCarouselOffset(track, offset, instant) {
if (instant) track.classList.add('is-initializing');
track.scrollLeft = offset * track.clientWidth;
if (instant) {
requestAnimationFrame(() => track.classList.remove('is-initializing'));
}
}

function initializeCarousel(track) {
const slides = getCarouselSlides(track);
if (slides.length < 2 || track.dataset.loopReady) return;

const count = slides.length;
const firstClone = slides[0].cloneNode(true);
const lastClone = slides[count - 1].cloneNode(true);
[firstClone, lastClone].forEach(clone => {
clone.dataset.loopClone = 'true';
clone.setAttribute('aria-hidden', 'true');
clone.querySelectorAll('a, button, input, select, textarea, [tabindex]').forEach(element => {
element.setAttribute('tabindex', '-1');
});
clone.querySelectorAll('img').forEach(image => image.alt = '');
});

track.prepend(lastClone);
track.append(firstClone);
track.dataset.realSlideCount = String(count);
track.dataset.loopReady = 'true';

const container = track.closest('.carousel-container');
const title = container.closest('.project-row').querySelector('.project-title').textContent.trim();
track.id = track.id || 'carousel-' + Math.random().toString(36).slice(2, 9);
track.setAttribute('role', 'region');
track.setAttribute('aria-roledescription', 'carousel');
track.setAttribute('aria-label', title + ' project images');
container.querySelectorAll('.carousel-btn').forEach(button => {
const direction = button.classList.contains('prev') ? 'Previous' : 'Next';
button.setAttribute('aria-label', direction + ' ' + title + ' image');
button.setAttribute('aria-controls', track.id);
button.removeAttribute('onclick');
button.addEventListener('click', () => scrollCarousel(button, button.classList.contains('prev') ? -1 : 1));
});
slides.forEach((slide, index) => {
slide.setAttribute('role', 'group');
slide.setAttribute('aria-roledescription', 'slide');
slide.setAttribute('aria-label', (index + 1) + ' of ' + count);
});

let settleTimer;
track.addEventListener('scroll', () => {
updateDots(track);
clearTimeout(settleTimer);
settleTimer = setTimeout(() => {
const rawIndex = Math.round(track.scrollLeft / track.clientWidth);
if (rawIndex === 0) setCarouselOffset(track, count, true);
else if (rawIndex === count + 1) setCarouselOffset(track, 1, true);
}, 160);
}, { passive: true });

setCarouselOffset(track, 1, true);
updateDots(track);
}

function scrollCarousel(btn, direction) {
const track = btn.closest('.carousel-container').querySelector('.carousel-track');
const count = Number(track.dataset.realSlideCount || getCarouselSlides(track).length);
if (count < 2) return;

const currentIndex = getCarouselIndex(track);
const targetIndex = (currentIndex + direction + count) % count;
const targetOffset = direction > 0 && currentIndex === count - 1
? count + 1
: direction < 0 && currentIndex === 0
? 0
: targetIndex + 1;
const reduceMotion = document.documentElement.classList.contains('reduce-motion');
setCarouselOffset(track, targetOffset, reduceMotion);
updateDots(track);
scheduleCarousel(track, carouselRotationInterval);
}

function updateDots(track) {
const dots = track.closest('.carousel-container').querySelectorAll('.dot-indicator');
const currentIndex = getCarouselIndex(track);
dots.forEach((dot, index) => {
const active = index === currentIndex;
dot.style.background = active ? 'var(--spring-green)' : 'rgba(255, 255, 255, 0.3)';
dot.setAttribute('aria-current', active ? 'true' : 'false');
});
}

function clearCarouselTimer(track) {
clearTimeout(carouselTimers.get(track));
carouselTimers.delete(track);
}

function refreshCarouselAutoplay(delay) {
carouselTracks.forEach(track => {
if (canAutoRotate(track)) scheduleCarousel(track, delay);
else clearCarouselTimer(track);
});
}

function canAutoRotate(track) {
const container = track.closest('.carousel-container');
return carouselAutoRotateEnabled
&& !document.documentElement.classList.contains('reduce-motion')
&& !document.hidden
&& !container.matches(':hover')
&& !container.contains(document.activeElement);
}

function scheduleCarousel(track, delay) {
clearCarouselTimer(track);
if (!canAutoRotate(track)) return;
carouselTimers.set(track, setTimeout(() => {
if (!canAutoRotate(track)) return;
const count = Number(track.dataset.realSlideCount);
const currentIndex = getCarouselIndex(track);
const targetOffset = currentIndex === count - 1 ? count + 1 : currentIndex + 2;
const reduceMotion = document.documentElement.classList.contains('reduce-motion');
setCarouselOffset(track, targetOffset, reduceMotion);
updateDots(track);
scheduleCarousel(track, carouselRotationInterval);
}, delay));
}

carouselTracks.forEach(track => {
initializeCarousel(track);
const container = track.closest('.carousel-container');
container.addEventListener('mouseenter', () => clearCarouselTimer(track));
container.addEventListener('mouseleave', () => scheduleCarousel(track, carouselRotationInterval));
container.addEventListener('pointerdown', () => clearCarouselTimer(track));
container.addEventListener('pointerup', () => scheduleCarousel(track, carouselRotationInterval));
container.addEventListener('focusin', () => clearCarouselTimer(track));
container.addEventListener('focusout', event => {
if (!container.contains(event.relatedTarget)) scheduleCarousel(track, carouselRotationInterval);
});
});

document.addEventListener('visibilitychange', () => {
carouselTracks.forEach(track => {
if (document.hidden) clearCarouselTimer(track);
else scheduleCarousel(track, carouselRotationInterval);
});
});
